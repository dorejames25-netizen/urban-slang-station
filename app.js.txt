// GLOBAL STATE CONTRACT
const State = {
  currentMood: 'CYBER_PLASMA',
  activePage: 'dashboard',
  intakeData: {
    title: 'London Pub Chaos',
    age: '18+',
    count: 100,
    swearLevel: 2,
    idea: ''
  },
  generatedCards: [],
  currentBatchIndex: 0
};

document.addEventListener('DOMContentLoaded', () => {
  initNavigation();
  initMoodEngine();
  initIntakeControls();
  initCompiler();
});

// ROUTING & NAVIGATION
function initNavigation() {
  const navButtons = document.querySelectorAll('.nav-btn');
  navButtons.forEach(btn => {
    btn.addEventListener('click', (e) => {
      const targetView = e.target.getAttribute('data-target');
      switchPage(targetView);
    });
  });

  document.getElementById('heroStartBtn')?.addEventListener('click', () => {
    switchPage('intake');
  });
}

function switchPage(pageId) {
  State.activePage = pageId;
  document.querySelectorAll('.page-view').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));

  const activeView = document.getElementById(pageId);
  if (activeView) activeView.classList.add('active');

  const navBtn = document.querySelector(`.nav-btn[data-target="${pageId}"]`);
  if (navBtn) navBtn.classList.add('active');

  if (pageId === 'matrix') {
    renderMatrixBatch();
  }
}

// MOOD THEME ENGINE
function initMoodEngine() {
  const selector = document.getElementById('moodSelect');
  selector?.addEventListener('change', (e) => {
    setMoodDirect(e.target.value);
  });
}

function setMoodDirect(moodKey) {
  State.currentMood = moodKey;
  document.documentElement.setAttribute('data-mood', moodKey);
  
  const badge = document.getElementById('currentMoodBadge');
  if (badge) badge.textContent = moodKey;

  const selector = document.getElementById('moodSelect');
  if (selector) selector.value = moodKey;
}

// INTAKE CONTROLS
function initIntakeControls() {
  const ageBtns = document.querySelectorAll('.segment-btn');
  ageBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      ageBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      State.intakeData.age = btn.getAttribute('data-age');
    });
  });

  const swearBtns = document.querySelectorAll('.swear-level-btn');
  swearBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      swearBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      State.intakeData.swearLevel = parseInt(btn.getAttribute('data-level'));
    });
  });

  document.getElementById('proceedToCompilerBtn')?.addEventListener('click', () => {
    State.intakeData.title = document.getElementById('gameTitle').value || 'Untitled Game';
    State.intakeData.count = parseInt(document.getElementById('cardCount').value);
    State.intakeData.idea = document.getElementById('gameIdea').value;
    
    document.getElementById('intakeJsonSummary').textContent = JSON.stringify(State.intakeData, null, 2);
    switchPage('compiler');
  });
}

// COMPILER ENGINE
function initCompiler() {
  document.getElementById('consultClaudeBtn')?.addEventListener('click', () => {
    const claudeOutput = document.getElementById('claudeOutput');
    claudeOutput.value = "Consulting Claude AI...\nCompiling structured card schema for " + State.intakeData.count + " cards...";
    
    setTimeout(() => {
      State.generatedCards = generateMockCards(State.intakeData.count);
      claudeOutput.value = JSON.stringify({
        game: State.intakeData.title,
        cards: State.generatedCards.slice(0, 3) // Preview slice
      }, null, 2);
      
      document.getElementById('sendToMatrixBtn')?.removeAttribute('disabled');
    }, 1000);
  });

  document.getElementById('sendToMatrixBtn')?.addEventListener('click', () => {
    switchPage('matrix');
  });
}

function generateMockCards(total) {
  const types = ['Truth', 'Dare', 'Challenge', 'Pub Rule', 'Voting'];
  const cards = [];
  
  for (let i = 1; i <= total; i++) {
    cards.push({
      id: `CARD-${String(i).padStart(3, '0')}`,
      type: types[Math.floor(Math.random() * types.length)],
      text: `Card ${i}: Reveal your most embarrassing pub story or perform 5 pushups!`,
      round: Math.ceil(i / (total / 5))
    });
  }
  return cards;
}
